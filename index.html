Hazırladığınız görsel linki ve seslendirme metnini doğrudan indirilebilir MP4 dosyasına dönüştürmek için aşağıdaki Colab kodunu kullanabilirsiniz:

import asyncio, os, requests, edge_tts, gradio as gr
from PIL import Image, ImageDraw, ImageFont
import numpy as np
from moviepy.editor import ImageClip, AudioFileClip, concatenate_videoclips

async def sahne_mp4_uret(gorsel_url, metin, ses="tr-TR-AhmetNeural"):
    await edge_tts.Communicate(metin.strip(), ses, rate="+5%").save("ses.mp3")
    ses_clip = AudioFileClip("ses.mp3")
    sure = ses_clip.duration + 0.3

    img_data = requests.get(gorsel_url.strip()).content
    with open("sahne.jpg", "wb") as f: f.write(img_data)
    im = Image.open("sahne.jpg").convert("RGB").resize((1080, 1920))
    draw = ImageDraw.Draw(im)
    draw.rectangle([(60, 1450), (1020, 1680)], fill=(0, 0, 0))
    draw.text((100, 1520), metin[:65], fill=(255, 230, 0))
    im.save("kare.jpg")

    video = ImageClip("kare.jpg").set_duration(sure).set_audio(ses_clip)
    video.write_videofile("cikti_reels.mp4", fps=24, codec="libx264", audio_codec="aac")
    return "cikti_reels.mp4"

with gr.Blocks() as demo:
    u, m, s = gr.Textbox(label="Görsel URL"), gr.Textbox(label="Metin"), gr.Textbox(value="tr-TR-AhmetNeural")
    gr.Button("MP4 Oluştur").click(fn=sahne_mp4_uret, inputs=[u, m, s], outputs=gr.Video(), api_name="mp4_uret")

demo.launch(share=True)
